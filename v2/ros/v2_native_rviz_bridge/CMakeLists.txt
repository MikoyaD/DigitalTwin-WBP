cmake_minimum_required(VERSION 3.0.2)
project(v2_native_rviz_bridge)
find_package(catkin REQUIRED COMPONENTS roscpp rviz pluginlib moveit_core moveit_ros_visualization moveit_msgs)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
catkin_package()
include_directories(${catkin_INCLUDE_DIRS})
add_library(v2_native_rviz_bridge SHARED src/display.cpp)
target_link_libraries(v2_native_rviz_bridge ${catkin_LIBRARIES} Qt5::Widgets)
set_target_properties(v2_native_rviz_bridge PROPERTIES CXX_STANDARD 14 LIBRARY_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/lib)
